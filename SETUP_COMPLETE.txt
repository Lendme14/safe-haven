💡 QUICK CODE EXAMPLES
──────────────────────────────────────────────────────────────────────────────

AUDIO RECORDING:
────────────────
import { useAudioRecording } from '@/hooks/useCapacitor';

const MyComponent = () => {
  const { isRecording, startRecording, stopRecording } = useAudioRecording();
  
  return (
    <>
      <button onClick={startRecording}>Start</button>
      <button onClick={stopRecording}>Stop</button>
      {isRecording && <p>Recording...</p>}
    </>
  );
};

BIOMETRIC AUTH:
───────────────
import { useBiometricAuth } from '@/hooks/useCapacitor';

const AuthComponent = () => {
  const { isAvailable, authenticate } = useBiometricAuth();
  
  return (
    <>
      {isAvailable && (
        <button onClick={authenticate}>Authenticate</button>
      )}
    </>
  );
};

NOTIFICATIONS:
──────────────
import { useNotifications } from '@/hooks/useCapacitor';

const AlertComponent = () => {
  const { showPersistentNotification } = useNotifications();
  
  const handleAlert = async () => {
    await showPersistentNotification({
      title: 'Alert',
      body: 'Important message'
    });
  };
  
  return <button onClick={handleAlert}>Show Alert</button>;
};
